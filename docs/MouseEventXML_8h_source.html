216inline
217DRAIN_XML_TREE(N) & MouseEventXML<N>::getListenerInitSubScope() const {
218 DRAIN_XML_TREE(N) & scope = getListenerInitScope(root, initName);
219 DRAIN_XML_TREE(N) & subScope = scope[N::tag_t::SCOPE_CURLY](N::tag_t::SCOPE_CURLY);
220 if (subScope.empty()){
221 subScope->setText("with(listener.ctx)");
222 subScope.addChild()->setProgramComment("Scope exposed by ", __FUNCTION__);
223 }
224 return subScope;
225}

237template <class N>
238inline
239void MouseEventXML<N>::connectClass(const drain::ClassXML & cls) const {
240 Logger mout(__FILE__, __FUNCTION__);
241
242 mout.debug("connecting CLASS ", cls, " = ", cls.strPrefixed());
243 DRAIN_XML_TREE(N) & scope = getListenerInitScope(root, initName);
244 scope[cls]->setText("listener.ctx['", cls, "'] = group.querySelector('", cls.strPrefixed(), "')");
245 connectedObjects.insert(cls);
