98 virtual inline
99 drain::Fuzzifier<double> & getFuzzifierZDR(LocalFunctorBank & bank) const override {
100 typedef drain::FuzzyTriangle<double> ftor_t;
101 ftor_t & functor = bank.clone<ftor_t>();
102 functor.set(::abs(zdrAbsThreshold), -::abs(zdrAbsThreshold), 0.0); // INVERSE
103 return functor;
104 }
105
107 virtual inline
108 drain::Fuzzifier<double> & getFuzzifierRHOHV(LocalFunctorBank & bank) const override {
109 typedef drain::FuzzyStep<double> ftor_t;
110 ftor_t & functor = bank.clone<ftor_t>();
111 // Ensure INVERSE (decreasing step function)
112 if (rhoHVthreshold.min > rhoHVthreshold.max){
113 functor.set(rhoHVthreshold);
114 }
115 else if (rhoHVthreshold.min < rhoHVthreshold.max){
116 functor.set(rhoHVthreshold.max, rhoHVthreshold.min); // invert
117 }
118 else {
119 functor.set(rhoHVthreshold.max, 0.8 * rhoHVthreshold.max);
120 }
121 return functor;
122 }
