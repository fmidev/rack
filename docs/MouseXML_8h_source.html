199template <class N>
200void MouseXML::adjustListenerGeoRef(DRAIN_XML_TREE(N) & handlerScopeJS, CoordFlagger::ivalue_t coords) const {
201
202 const CoordFlagger flags(coords);
203
204 DRAIN_XML_TREE(N) & coordScopeJS = handlerScopeJS(COORD_INIT_SCOPE);
205
206 if (flags.isSet(IMAGE_COORDS)){
207 coordScopeJS[flags.str()] -> setProgramComment("Coord request: ", flags.str());
208 // TODO: detect if MOVE has already been defined, and drop (duplicated) coordinate processing from here.
209 coordScopeJS["ctx.bbox"] = "ctx.bbox = evt.target.getBoundingClientRect();";
210 coordScopeJS["COORDS"] -> setProgramComment("Image coordinates");
211 coordScopeJS["ctx.x"] = "ctx.x = Math.floor(evt.clientX - ctx.bbox.left);";
212 coordScopeJS["ctx.y"] = "ctx.y = Math.floor(evt.clientY - ctx.bbox.top);";
213 if (flags.isSet(RELATIVE_COORDS)){
214 coordScopeJS["RELATIVE_COORDS"] -> setProgramComment("Relative coordinates");
215 coordScopeJS["ctx.rx"] = "ctx.rx = ctx.x / ctx.bbox.width;";
216 coordScopeJS["ctx.ry"] = "ctx.ry = ctx.y / ctx.bbox.height;";
217 if (flags.isSet(GEOGRAPHIC_COORDS)){
218 coordScopeJS["GEOGRAPHIC_COORDS"] -> setProgramComment("Geographic coordinates");
219 // geo_bbox.left + rx*geo_bbox.width
220 coordScopeJS["ctx.gx"] = "ctx.gx = ctx.georef.bbox.left + ctx.rx*ctx.georef.bbox.width;";
221 coordScopeJS["ctx.gy"] = "ctx.gy = ctx.georef.bbox.top + ctx.ry*ctx.georef.bbox.height;";
222 }
223 }
224 }
